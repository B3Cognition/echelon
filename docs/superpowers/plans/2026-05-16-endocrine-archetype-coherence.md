# Endocrine archetype coherence — implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bring the endocrine subsystem's archetype assignments and runtime modifier in line with each agent's actual role, make every dispatched agent receive role-appropriate interpretation guidance instead of a generic urgency phrase, and add a single test that catches future roster / archetype / interpretation drift.

**Architecture:** Single source of truth in `echelon-config.yml`: extend the existing `endocrine.baselines:` block with a sibling `endocrine.interpretations:` block. `endocrine.sh` reads it at runtime via the existing `yaml_get` helper and emits a richer multi-line `[ENDOCRINE — <archetype> archetype]` modifier per dispatch. Seven agents move to archetypes that match their roles. Every agent `.md` file gains one standardized line in its `## Role` section pointing at the runtime block. A new consistency test asserts the four-way coherence (disk ↔ ALL_AGENTS ↔ archetype map ↔ config baselines+interpretations).

**Tech Stack:** Bash 4+, jq, Python 3 (for YAML parsing — already used in endocrine.sh), PyYAML, plain-bash tests (existing convention in `tests/unit/test-endocrine-*.sh`).

**Spec reference:** `docs/superpowers/specs/2026-05-16-endocrine-archetype-coherence-design.md` (sections 1–5).

Replace the entire `cmd_get_full_prompt_modifier()` function (from `cmd_get_full_prompt_modifier() {` through the matching closing `}` at the end of the function) with:

```bash
cmd_get_full_prompt_modifier() {
  local agent="${1:?Usage: endocrine.sh get_full_prompt_modifier <agent>}"
  require_jq

  # Read all 6 hormone values
  local adr dop cor ser oxy nor
  adr=$(read_hormone_value "$agent" "adrenaline")     || return 1
  dop=$(read_hormone_value "$agent" "dopamine")       || return 1
  cor=$(read_hormone_value "$agent" "cortisol")       || return 1
  ser=$(read_hormone_value "$agent" "serotonin")      || return 1
  oxy=$(read_hormone_value "$agent" "oxytocin")       || return 1
  nor=$(read_hormone_value "$agent" "norepinephrine") || return 1

  # Classify each hormone: LOW <= 0.25, MEDIUM (0.25, 0.75), HIGH >= 0.75
  # (Spec section 2 levels.)
  classify() {
    awk -v v="$1" 'BEGIN {
      if (v <= 0.25) print "low"
      else if (v >= 0.75) print "high"
      else print "medium"
    }'
  }

  local adr_lvl dop_lvl cor_lvl ser_lvl oxy_lvl nor_lvl
  adr_lvl=$(classify "$adr")
  dop_lvl=$(classify "$dop")
  cor_lvl=$(classify "$cor")
  ser_lvl=$(classify "$ser")
  oxy_lvl=$(classify "$oxy")
  nor_lvl=$(classify "$nor")

  local archetype
  archetype=$(agent_to_archetype "$agent")

  # Load summary + overlays from echelon-config.yml interpretations block.
  # Use yaml.safe_load via python3 — yaml_get only handles flat keys.
  local SUMMARY OVERLAYS_BLOB
  local INTERP_RAW
  if [[ -f "$REPO_ROOT/extension/echelon-config.yml" ]]; then
    CFG_PATH="$REPO_ROOT/extension/echelon-config.yml"
  elif [[ -f "$REPO_ROOT/echelon-config.yml" ]]; then
    CFG_PATH="$REPO_ROOT/echelon-config.yml"
  else
    CFG_PATH=""
  fi

  if [[ -n "$CFG_PATH" ]] && command -v $PYTHON &>/dev/null; then
    INTERP_RAW=$($PYTHON -c "
import yaml, sys
try:
    cfg = yaml.safe_load(open('$CFG_PATH')) or {}
except Exception:
    sys.exit(0)
arch = cfg.get('endocrine', {}).get('interpretations', {}).get('$archetype', {})
if not arch:
    sys.exit(0)
summary = (arch.get('summary') or '').strip()
overlays = arch.get('overlays', {}) or {}
print('---SUMMARY---')
print(summary)
print('---OVERLAYS---')
for k, v in overlays.items():
    print(f'{k}::{v}')
" 2>/dev/null)
  fi

  # If no interpretation block found, fall back to legacy single-line modifier.
  if [[ -z "$INTERP_RAW" ]]; then
    echo "[ENDOCRINE: all hormones MEDIUM] Normal operating conditions."
    return 0
  fi

  SUMMARY=$(echo "$INTERP_RAW" | sed -n '/^---SUMMARY---$/,/^---OVERLAYS---$/p' | sed '1d;$d')
  OVERLAYS_BLOB=$(echo "$INTERP_RAW" | sed -n '/^---OVERLAYS---$/,$p' | tail -n +2)

  # Emit the multi-line block.
  fmt() { printf "%.2f" "$1" 2>/dev/null || echo "$1"; }
  local adr_u dop_u cor_u ser_u oxy_u nor_u
  adr_u=$(echo "$adr_lvl" | tr '[:lower:]' '[:upper:]')
  dop_u=$(echo "$dop_lvl" | tr '[:lower:]' '[:upper:]')
  cor_u=$(echo "$cor_lvl" | tr '[:lower:]' '[:upper:]')
  ser_u=$(echo "$ser_lvl" | tr '[:lower:]' '[:upper:]')
  oxy_u=$(echo "$oxy_lvl" | tr '[:lower:]' '[:upper:]')
  nor_u=$(echo "$nor_lvl" | tr '[:lower:]' '[:upper:]')

  printf '[ENDOCRINE — %s archetype]\n' "$archetype"
  printf '  adrenaline: %s (%s)   dopamine: %s (%s)\n' "$(fmt "$adr")" "$adr_u" "$(fmt "$dop")" "$dop_u"
  printf '  cortisol:   %s (%s)   serotonin: %s (%s)\n' "$(fmt "$cor")" "$cor_u" "$(fmt "$ser")" "$ser_u"
  printf '  oxytocin:   %s (%s)   norepinephrine: %s (%s)\n' "$(fmt "$oxy")" "$oxy_u" "$(fmt "$nor")" "$nor_u"
  printf '\n'
  printf 'Interpretation (%s archetype):\n' "$archetype"
  echo "$SUMMARY" | sed 's/^/  /'

  # Emit overlays for any hormone at HIGH/LOW that has a matching overlay.
  emit_overlay() {
    local hormone="$1" level="$2"
    local key="${hormone}_${level}"
    local val
    val=$(echo "$OVERLAYS_BLOB" | sed -n "s|^${key}::||p")
    if [[ -n "$val" ]]; then
      local hormone_uc level_uc
      hormone_uc=$(echo "$hormone" | tr '[:lower:]' '[:upper:]')
      level_uc=$(echo "$level" | tr '[:lower:]' '[:upper:]')
      printf '  - %s %s: %s\n' "$level_uc" "$hormone_uc" "$val"
    fi
  }

  for pair in "adrenaline:$adr_lvl" "dopamine:$dop_lvl" "cortisol:$cor_lvl" \
              "serotonin:$ser_lvl" "oxytocin:$oxy_lvl" "norepinephrine:$nor_lvl"; do
    hormone="${pair%%:*}"
    level="${pair##*:}"
    [[ "$level" == "medium" ]] && continue
    emit_overlay "$hormone" "$level"
  done

  # Preserve the existing [CALIBRATION] suffix — commander.md §198 reads it from
  # this function's output and appends it to the agent's prompt. Removing it
  # would break the protocol. (Future refactor: deduplicate against commander.md
  # §175-190's own calibration injection — out of scope for this design.)
  local SCORES_FILE="$REPO_ROOT/knowledge-base/agent-scores.yaml"
  if [[ -f "$SCORES_FILE" ]] && command -v $PYTHON &>/dev/null; then
    $PYTHON -c "
import yaml, sys
try:
    with open('$SCORES_FILE') as f:
        data = yaml.safe_load(f) or {}
    agents = data.get('agents', {})
    a = agents.get('$agent', {})
    history = a.get('history', a.get('run_history', []))
    if not history:
        sys.exit(0)
    last = history[-1] if isinstance(history, list) else {}
    score = last.get('quality_score', last.get('score', '?'))
    modes = last.get('failure_modes', [])
    if modes:
        fm = modes[0]
        print(f'[CALIBRATION] Prior score: {score}. Miss: {fm.get(\"type\",\"unknown\")} ({fm.get(\"count\",\"?\")}x). Example: {fm.get(\"example\",\"none\")}')
    else:
        print(f'[CALIBRATION] Prior score: {score}. No failure modes recorded.')
except Exception as e:
    print(f'[CALIBRATION] COLD START — error reading scores: {e}')
" 2>/dev/null
  fi
}
```

- Reads archetype via `agent_to_archetype`.

- **Preserves** the `[CALIBRATION]` suffix at the end so commander.md §198's protocol still works. (Spec doesn't mandate removing it; the existing CALIBRATION suffix is orthogonal to Section 2's multi-line format.)

```bash
#!/usr/bin/env bash
# agent-endocrine-rewire.sh — Idempotent inserter for the endocrine-awareness
# marker line in every agent .md file under extension/agents/<layer>/.
#
# Usage:
#   bash agent-endocrine-rewire.sh [--dry-run] [--remove]
#
# Inserts (after each agent's "## Role" body, before the next "##" heading):
#
#   > **Endocrine awareness.** Your dispatched context pack includes an
#   > `[ENDOCRINE]` block from `endocrine.sh get_full_prompt_modifier`:
#   > your current hormone levels … plus role-appropriate interpretation
#   > from your archetype. It's not narration — it's behavior modulation.
#   > Read and act on it before producing output.
#
# Idempotent: re-running on a file that already contains the marker is a no-op.
#
# Exit codes:
#   0 = success (or dry-run completed)
#   1 = called outside a project root, OR a file lacks "## Role" heading

set -euo pipefail

DRY_RUN=false
REMOVE=false
while [[ $# -gt 0 ]]; do
  case "$1" in
    --dry-run) DRY_RUN=true; shift ;;
    --remove)  REMOVE=true; shift ;;
    -h|--help)
      sed -n '2,18p' "$0" | sed 's/^# \{0,1\}//'
      exit 0
      ;;
    *) echo "agent-endocrine-rewire: unknown arg: $1" >&2; exit 1 ;;
  esac
done

ROOT="$(pwd)"
while [ "$ROOT" != "/" ] && [ ! -d "$ROOT/.specify" ]; do
  ROOT=$(dirname "$ROOT")
done
if [ "$ROOT" = "/" ]; then
  echo "agent-endocrine-rewire: no .specify/ in CWD or parents" >&2
  exit 1
fi
cd "$ROOT"

MARKER="**Endocrine awareness.**"
read -r -d '' BLOCK <<'EOF' || true
> **Endocrine awareness.** Your dispatched context pack includes an `[ENDOCRINE]` block from `endocrine.sh get_full_prompt_modifier`: your current hormone levels (adrenaline, dopamine, cortisol, serotonin, oxytocin, norepinephrine) plus role-appropriate interpretation from your archetype. It's not narration — it's behavior modulation. Read and act on it before producing output.
EOF

# Three agent-file directories to process (source + deployed copies).
DIRS=(
  "extension/agents"
  ".specify/extensions/echelon/agents"
)

processed=0
inserted=0
skipped=0
removed=0
errors=0

for base in "${DIRS[@]}"; do
  [ -d "$base" ] || { echo "  (skip — $base not present)"; continue; }
  while IFS= read -r -d '' f; do
    processed=$((processed + 1))

    if $REMOVE; then
      if grep -qF "$MARKER" "$f"; then
        if $DRY_RUN; then
          echo "  WOULD REMOVE marker in $f"
        else
          # Delete the entire blockquote line(s) containing the marker.
          python3 - <<PY
import sys, re
path = "$f"
text = open(path).read()
# Match: optional blank line + the blockquote line containing the marker + optional blank line
new = re.sub(r'\n\n> \*\*Endocrine awareness\.\*\*[^\n]*\n', '\n', text, count=1)
open(path, 'w').write(new)
PY
        fi
        removed=$((removed + 1))
      else
        skipped=$((skipped + 1))
      fi
      continue
    fi

    if grep -qF "$MARKER" "$f"; then
      skipped=$((skipped + 1))
      continue
    fi

    # Find "## Role" line; insert BLOCK before the next "## " heading (or EOF).
    if ! grep -qE '^## [Rr]ole' "$f"; then
      echo "  ERROR — $f has no '## Role' heading" >&2
      errors=$((errors + 1))
      continue
    fi

    if $DRY_RUN; then
      echo "  WOULD INSERT marker in $f"
      inserted=$((inserted + 1))
      continue
    fi

    python3 - <<PY
import re
path = "$f"
block = """
$BLOCK
"""
text = open(path).read()
# Find ## Role heading, scan forward for next "## " heading (any case after #), or EOF.
m = re.search(r'^## [Rr]ole.*?$', text, flags=re.M)
if not m:
    raise SystemExit("no role heading after grep matched — internal error")
start = m.end()
# Find next H2 after the Role body
next_h2 = re.search(r'^## ', text[start:], flags=re.M)
insertion_pos = start + (next_h2.start() if next_h2 else len(text) - start)
new_text = text[:insertion_pos].rstrip() + "\n\n" + block.strip() + "\n\n" + text[insertion_pos:].lstrip()
open(path, "w").write(new_text)
PY
    inserted=$((inserted + 1))
  done < <(find "$base" -type f -name "*.md" -print0)
done

echo
echo "agent-endocrine-rewire summary:"
echo "  processed: $processed"
echo "  inserted:  $inserted"
echo "  removed:   $removed"
echo "  skipped:   $skipped  (already had marker, or didn't have one to remove)"
echo "  errors:    $errors"
exit $((errors == 0 ? 0 : 1))
```

```bash
#!/usr/bin/env bash
# Unit test — Section-5 consistency validator.
# Asserts the four-way coherence:
#   1. ALL_AGENTS ↔ agent_to_archetype (every agent resolves non-default)
#   2. agent_to_archetype outputs ↔ baselines (every archetype has baseline)
#   3. baselines ↔ interpretations (every baseline has matching interpretation)
#   4. ALL_AGENTS ↔ disk (every agent listed has a .md file)
#   5. disk ↔ ALL_AGENTS (every .md file is listed)
#   6. disk ↔ marker (every .md file has the endocrine-awareness blockquote)

set -uo pipefail

REPO_ROOT="$(CDPATH='' cd "$(dirname "$0")/../.." && pwd)"
ENDOCRINE="$REPO_ROOT/extension/scripts/bash/endocrine.sh"
CONFIG="$REPO_ROOT/extension/echelon-config.yml"
AGENTS_DIR="$REPO_ROOT/extension/agents"
MARKER="**Endocrine awareness.**"

# shellcheck disable=SC1090
source "$ENDOCRINE"

pass=0
fail=0
section() { printf "\n[%d] %s\n" "$1" "$2"; }
check() {
  local desc="$1" cond="$2"
  if eval "$cond"; then pass=$((pass+1)); printf "  PASS  %s\n" "$desc"
  else fail=$((fail+1)); printf "  FAIL  %s\n" "$desc"; fi
}

# Build archetype set from agent_to_archetype outputs.
ARCH_SET=$(for a in "${ALL_AGENTS[@]}"; do agent_to_archetype "$a"; done | sort -u)

# Parse baselines + interpretations archetypes from YAML.
BASELINE_ARCHS=$(python3 -c "
import yaml
d = yaml.safe_load(open('$CONFIG')) or {}
for k in sorted((d.get('endocrine',{}).get('baselines',{}) or {}).keys()):
    print(k)
")
INTERP_ARCHS=$(python3 -c "
import yaml
d = yaml.safe_load(open('$CONFIG')) or {}
for k in sorted((d.get('endocrine',{}).get('interpretations',{}) or {}).keys()):
    print(k)
")

# Disk roster.
DISK_AGENTS=$(find "$AGENTS_DIR" -type f -name '*.md' -exec basename {} .md \; | tr '[:lower:]-' '[:upper:]_' | sort -u)
ALL_AGENTS_UPPER=$(printf '%s\n' "${ALL_AGENTS[@]}" | sort -u)

# Assertion 1: every ALL_AGENTS entry has a non-default archetype.
section 1 "ALL_AGENTS → archetype (no silent default-control)"
default_hits=0
for a in "${ALL_AGENTS[@]}"; do
  # Default-fallback agents would be those NOT in any case branch. We detect by
  # checking if any agent name appears literally in the case statement.
  if ! grep -qE "[|(]${a}[|)]" "$ENDOCRINE"; then
    default_hits=$((default_hits + 1))
    echo "    silent default: $a (would fall through to control)"
  fi
done
check "every ALL_AGENTS entry has an explicit archetype case" "[ $default_hits -eq 0 ]"

# Assertion 2: every archetype returned has a baseline.
section 2 "archetype outputs ↔ baselines"
missing_baseline=0
for arch in $ARCH_SET; do
  if ! echo "$BASELINE_ARCHS" | grep -qx "$arch"; then
    missing_baseline=$((missing_baseline + 1))
    echo "    no baseline for archetype: $arch"
  fi
done
check "every archetype has a baseline" "[ $missing_baseline -eq 0 ]"

# Assertion 3: every baseline has an interpretation.
section 3 "baselines ↔ interpretations"
missing_interp=0
for arch in $BASELINE_ARCHS; do
  if ! echo "$INTERP_ARCHS" | grep -qx "$arch"; then
    missing_interp=$((missing_interp + 1))
    echo "    no interpretation for baseline: $arch"
  fi
done
check "every baseline has an interpretation" "[ $missing_interp -eq 0 ]"

# Assertion 4: every ALL_AGENTS entry has a disk file.
section 4 "ALL_AGENTS ↔ disk"
missing_disk=0
for a in "${ALL_AGENTS[@]}"; do
  if ! echo "$DISK_AGENTS" | grep -qx "$a"; then
    missing_disk=$((missing_disk + 1))
    echo "    ALL_AGENTS entry has no .md file: $a"
  fi
done
check "every ALL_AGENTS entry has an agent file" "[ $missing_disk -eq 0 ]"

# Assertion 5: every disk file is in ALL_AGENTS.
section 5 "disk ↔ ALL_AGENTS"
missing_in_list=0
for a in $DISK_AGENTS; do
  if ! echo "$ALL_AGENTS_UPPER" | grep -qx "$a"; then
    missing_in_list=$((missing_in_list + 1))
    echo "    disk file not in ALL_AGENTS: $a"
  fi
done
check "every agent file is in ALL_AGENTS" "[ $missing_in_list -eq 0 ]"

# Assertion 6: every agent file has the endocrine-awareness marker.
section 6 "disk → endocrine-awareness marker"
missing_marker=0
while IFS= read -r -d '' f; do
  if ! grep -qF "$MARKER" "$f"; then
    missing_marker=$((missing_marker + 1))
    echo "    no marker: $f"
  fi
done < <(find "$AGENTS_DIR" -type f -name '*.md' -print0)
check "every agent file contains the marker" "[ $missing_marker -eq 0 ]"

echo
echo "Pass: $pass / 6  Fail: $fail"
exit $((fail == 0 ? 0 : 1))
```
